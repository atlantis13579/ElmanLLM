cmake_minimum_required(VERSION 3.24)
get_filename_component(ELMAN_BUILD_VARIANT "${CMAKE_BINARY_DIR}" NAME)
if(CMAKE_GENERATOR MATCHES "Visual Studio")
    list(APPEND CMAKE_VS_GLOBALS
        "ElmanIntermediateRoot=${CMAKE_CURRENT_LIST_DIR}/Intermediate/${ELMAN_BUILD_VARIANT}"
        "ForceImportBeforeCppTargets=${CMAKE_CURRENT_LIST_DIR}/CMake/OutputDirectories.targets")
endif()
project(Elman VERSION 0.1.0 LANGUAGES CXX)

option(ELMAN_ENABLE_CUDA "Build the native CUDA backend" OFF)
option(ELMAN_ENABLE_COMPUTE "Build the native Direct3D 11 compute backend" ${WIN32})
option(ELMAN_ENABLE_METAL "Build the native Apple Metal backend" ${APPLE})
option(ELMAN_BUILD_TESTS "Build numerical and integration tests" ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/Binaries/${CMAKE_SYSTEM_NAME}/${ELMAN_BUILD_VARIANT}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/Binaries/${CMAKE_SYSTEM_NAME}/${ELMAN_BUILD_VARIANT}")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${PROJECT_SOURCE_DIR}/Binaries/${CMAKE_SYSTEM_NAME}/${ELMAN_BUILD_VARIANT}")
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
file(MAKE_DIRECTORY "${PROJECT_SOURCE_DIR}/Intermediate")
if(ELMAN_BUILD_TESTS)
    enable_testing()
endif()
# Keep generated projects in Build and compiled target files in Intermediate.
if(CMAKE_GENERATOR MATCHES "Ninja|Makefiles")
    add_subdirectory(CMake/Targets "${PROJECT_SOURCE_DIR}/Intermediate/${ELMAN_BUILD_VARIANT}/Targets")
else()
    add_subdirectory(CMake/Targets "${CMAKE_BINARY_DIR}/Targets")
endif()
